import os, subprocess

linije = []
for f in sorted(os.listdir("pdf")):
    if f.lower().endswith(".pdf"):
        txt = subprocess.check_output(["pdftotext", "-l", "1", os.path.join("pdf", f), "-"], stderr=subprocess.DEVNULL).decode("utf-8", errors="ignore")
        redovi = " | ".join(l.strip() for l in txt.splitlines() if l.strip())[:160]
        linije.append(f"{f:35s} -> {redovi}")

open("provjerasve.txt", "w", encoding="utf-8").write("\n".join(linije) + "\n")
print(f"Gotovo! Spremljeno {len(linije)} redaka u provjerasve.txt.")
